@code {
    private PagedResult<Beer> _pagedResult = null!;

    private readonly PageFilterSorting _filterAndSorting = new()
    {
        OrderBy = "Name"
    };


    //from querystring bier?brouwcode=682
    [Parameter] public int? BrewerId { get; set; }
    
    //From querystring bier?codeFromBrouwer=682
    [Parameter, SupplyParameterFromQuery] public int? CodeFromBrouwer { get; set; }
    
    [Parameter] public string? BrewerName { get; set; }

    [Parameter] public string? Country { get; set; }

    protected override void OnInitialized()
    {
        base.OnInitialized();
        
        var uriAsString = NavigationManager.Uri;
        var url = new Uri(uriAsString);
        var queryString = url.Query; // returns "?BrewerId=someInt"
        var queryParameters = HttpUtility.ParseQueryString(queryString);
        
        //Query parameters are always strings and the key (BrouwerId) is case sensitive!!
        var brouwcodeAsString = queryParameters["BrewerId"]; // returns "someInt"
        if (int.TryParse(brouwcodeAsString, out var brouwcode))
        {
            BrewerId = brouwcode;
        }
    }

    protected override void OnParametersSet()
    {
        base.OnParametersSet();
        LoadData(_filterAndSorting, 1, 10);
    }

    private void ApplyRouteParameters(PageFilterSorting pageFilterSorting)
    {
        if (BrewerId is not null)
        {
            pageFilterSorting.BrewerId = BrewerId;
        }
        else if (CodeFromBrouwer is not null)
        {
            pageFilterSorting.BrewerId = CodeFromBrouwer;
        }
        else if (BrewerName is not null)
        {
            pageFilterSorting.BrewerName = BrewerName;
            if (Country is not null)
            {
                pageFilterSorting.Country = Country;
            }
        }
    }

    private void LoadData(PageFilterSorting pageFilterSorting, int pageIndex, int pageSize)
    {
        ApplyRouteParameters(pageFilterSorting);
        _pagedResult = new BeerRepository().Get(pageFilterSorting, pageIndex, pageSize);
    }
    
    private void SortBy(string columnName)
    {
        if (_filterAndSorting.OrderBy == columnName && _filterAndSorting.OrderDirection == "asc")
        {
            _filterAndSorting.OrderBy = columnName;
            _filterAndSorting.OrderDirection = "desc";
        }
        else if (_filterAndSorting.OrderBy == columnName && _filterAndSorting.OrderDirection == "desc")
        {
            _filterAndSorting.OrderBy = columnName;
            _filterAndSorting.OrderDirection = "asc";
        }
        else
        {
            _filterAndSorting.OrderBy = columnName;
            _filterAndSorting.OrderDirection = "asc";
        }
        
        LoadData(_filterAndSorting, _pagedResult.Page, _pagedResult.PageSize);
    }

    private void GoToPage(int page)
    {
        LoadData(_filterAndSorting, page, _pagedResult.PageSize);
    }

    private void FilterByName(string? beerNameFilter)
    {
        _filterAndSorting.BeerName = beerNameFilter;
        _filterAndSorting.OrderBy = "name";
        _filterAndSorting.OrderDirection = "asc";
        LoadData(_filterAndSorting, _pagedResult.Page, _pagedResult.PageSize);
    }
}
